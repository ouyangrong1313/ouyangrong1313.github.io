查询 SQL

SELECT
  date AS stat_date,
  channel_sn,
  COUNT(*) AS motion_detection_event_count,
  AVG(CAST(preview_duration_ms AS DOUBLE)) AS avg_preview_duration_ms,
  MAX(CAST(preview_duration_ms AS DOUBLE)) AS max_preview_duration_ms,
  SUM(CAST(preview_duration_ms AS DOUBLE)) AS total_preview_duration_ms
FROM events
WHERE event = 'NvrMonitorFrameResult'
  AND date = '2026-09-11'
  AND channel_sn IN ({160 channel_sn values from the current default device list})
  AND recv_frame_time > 0
  AND subscribe_type = 1
GROUP BY date, channel_sn
ORDER BY total_preview_duration_ms DESC, channel_sn;
